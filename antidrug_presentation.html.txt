<script>
  function go(num) {
    document.querySelectorAll('.slide').forEach(s => s.classList.remove('active'));
    document.querySelector(`[data-id="${num}"]`).classList.add('active');
  }
  const answers = {
    1: false,
    2: false,
    3: false,
    4: true
  };
  function check(id, chosen) {
    const box = document.getElementById('ans' + id);
    if (chosen === answers[id]) {
      box.textContent = 'Верно!';
      box.className = 'feedback correct';
    } else {
      box.textContent = 'Неверно, это миф.';
      box.className = 'feedback wrong';
    }
  }
</script>
